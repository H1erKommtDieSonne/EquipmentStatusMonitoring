79 sqlite3_finalize(stmt);

86 r.is_faulty = sqlite3_column_int(stmt, 3) != 0;

88 out.push_back(std::move(r));
89 sqlite3_finalize(stmt);

198std::vector<JobRow> Engine::list_jobs(std::optional<JobStatus> status) const {
199 sqlite3* db = storage_.handle();
200 sqlite3_stmt* stmt = nullptr;
201
202 const char* SQL_ALL =
203 "SELECT id,device_address,worker_id,fault,status,created_at,started_at,finished_at "
204 "FROM jobs ORDER BY id;";
205 const char* SQL_FLT =
206 "SELECT id,device_address,worker_id,fault,status,created_at,started_at,finished_at "
207 "FROM jobs WHERE status=?1 ORDER BY id;";
208
209 if (sqlite3_prepare_v2(db, status ? SQL_FLT : SQL_ALL, -1, &stmt, nullptr) != SQLITE_OK)
210 throw std::runtime_error(std::string("prepare list_jobs failed: ") + sqlite3_errmsg(db));
211 if (status) sqlite3_bind_int(stmt, 1, static_cast<int>(*status));
212
213 std::vector<JobRow> out;
214 while (true) {
215 const int rc = sqlite3_step(stmt);
216 if (rc == SQLITE_ROW) {
217 JobRow r{};
218 r.id = sqlite3_column_int64(stmt, 0);
219 r.device_address = static_cast<std::uint32_t>(sqlite3_column_int64(stmt, 1));
220 if (sqlite3_column_type(stmt, 2) == SQLITE_NULL) r.worker_id = std::nullopt;
221 else r.worker_id = sqlite3_column_int64(stmt, 2);
222 r.fault = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 3));
223 r.status = static_cast<JobStatus>(sqlite3_column_int(stmt, 4));
224 r.created_at = sqlite3_column_int64(stmt, 5);
225 r.started_at = (sqlite3_column_type(stmt, 6) == SQLITE_NULL) ? std::nullopt
226 : std::optional<std::int64_t>(sqlite3_column_int64(stmt, 6));
227 r.finished_at = (sqlite3_column_type(stmt, 7) == SQLITE_NULL) ? std::nullopt
228 : std::optional<std::int64_t>(sqlite3_column_int64(stmt, 7));
229 out.push_back(std::move(r));
230 }
231 else if (rc == SQLITE_DONE) break;
232 else { std::string msg = sqlite3_errmsg(db); sqlite3_finalize(stmt); throw std::runtime_error("list_jobs failed: " + msg); }
233 }
234 sqlite3_finalize(stmt);
235 return out;
236}
