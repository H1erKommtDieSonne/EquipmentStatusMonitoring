84static py::list engine_list_jobs(const Engine& engine, py::object status_obj)
85{
86 std::optional<JobStatus> status;
87
88 if (!status_obj.is_none())
89 {
90 status = py::extract<JobStatus>(status_obj);
91 }
92
93 py::list result;
94 const std::vector<JobRow> jobs = engine.list_jobs(status);
95 for (const auto& j : jobs)
96 {
97 result.append(j);
98 }
99 return result;
