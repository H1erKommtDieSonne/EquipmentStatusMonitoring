273 def on_open_db(self) -> None:
274 db_path = self.db_path_var.get().strip()
275 if not db_path:
276 messagebox.showerror("Ошибка", "Путь к файлу БД не может быть пустым.")

292 def on_upsert_device(self) -> None:
293 engine = self._require_engine()
294 if engine is None:
295 return
296
297 name = self.dev_name_var.get().strip()
298 ipv4 = self.dev_ipv4_var.get().strip()
299 if not name or not ipv4:
300 messagebox.showerror("Ошибка", "Имя и IPv4 обязательны.")
301 return
302
303 try:
304 address = ipv4_to_u32(ipv4)
305 except Exception as exc:
306 messagebox.showerror("Ошибка", f"Неверный IPv4: {exc}")
307 return
308
309 try:
310 uptime = int(self.dev_uptime_var.get().strip() or "0")
311 standby = int(self.dev_standby_var.get().strip() or "0")
312 except ValueError:
313 messagebox.showerror("Ошибка", "Uptime и ожидание должны быть целыми.")
314 return
315
316 priority_enum = PRIORITY_CODE_TO_ENUM.get(
317 self.dev_priority_code_var.get(), getattr(esm.ServicePriority, "None")
318 )
319
320 is_faulty = bool(self.dev_is_faulty_var.get())
321 is_reserve = bool(self.dev_is_reserve_var.get())
322 fault_text = self.dev_fault_var.get().strip()
323
324 try:
325 if is_faulty:
326 device = esm.FaultyDeviceEx(
327 name, address, priority_enum, fault_text or "Unknown fault"
328 )
329 elif is_reserve:
330 device = esm.ReserveDevice(name, address, priority_enum, uptime, standby)
331 else:
332 device = esm.HealthyDevice(name, address, priority_enum, uptime)
333
334 engine.upsert_device(device)
335 except Exception as exc:
336 messagebox.showerror("Ошибка", f"Не удалось сохранить устройство: {exc}")
337 return
338
339 self.refresh_devices()

367 def on_breakdown(self) -> None:
368 engine = self._require_engine()
369 if engine is None:
370 return
371
372 ipv4 = self.breakdown_ipv4_var.get().strip()
373 fault = self.breakdown_fault_var.get().strip() or "Unknown fault"
374 if not ipv4:
375 messagebox.showerror("Ошибка", "IPv4 обязателен.")
376 return
377
378 try:
379 address = ipv4_to_u32(ipv4)
380 except Exception as exc:
381 messagebox.showerror("Ошибка", f"Неверный IPv4: {exc}")
382 return
383
384 try:
385 job_id = engine.breakdown(address, fault)
386 except Exception as exc:
387 messagebox.showerror("Ошибка", f"Не удалось создать задачу: {exc}")
388 return

390 self.refresh_jobs()

393 def refresh_workers(self) -> None:
394 engine = self.engine
395 if engine is None:
396 return
397
398 for item in self.workers_tree.get_children():
399 self.workers_tree.delete(item)
400
401 try:
402 workers = engine.list_workers()
403 except Exception:
404 return
405
406 for w in workers:
407 self.workers_tree.insert("", "end", values=(w.id, w.name, w.max_jobs, w.skill))

409 def refresh_devices(self) -> None:
410 engine = self.engine
411 if engine is None:
412 return
413
414 for item in self.devices_tree.get_children():
415 self.devices_tree.delete(item)
416
417 try:
418 devices = engine.list_devices()

428 values=(
429 ipv4,
430 d.name,
431 label,
432 "Да" if d.is_faulty else "Нет",

434 ),

437 def refresh_jobs(self) -> None:
438 engine = self.engine
439 if engine is None:
440 return
441
442 for item in self.jobs_tree.get_children():
443 self.jobs_tree.delete(item)
444
445 try:
446 jobs = engine.list_jobs()
447 except Exception:
448 return
449
450 for j in jobs:
451 device_val = j.device_address
452 try:
453 device_shown = u32_to_ipv4(int(device_val))
454 except Exception:
455 device_shown = str(device_val)
456
457 worker = j.worker_id if getattr(j, "worker_id", None) is not None else ""
458 status = j.status
459 self.jobs_tree.insert("", "end", values=(j.id, device_shown, worker, status, j.fault))
